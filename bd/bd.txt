
-- ===============================
-- 1️⃣ Tabla: rol
-- ===============================
CREATE TABLE rol (
  id_rol INT AUTO_INCREMENT PRIMARY KEY,
  nombre_rol VARCHAR(100) UNIQUE NOT NULL,
  descripcion TEXT
);

-- ===============================
-- 2️⃣ Tabla: permisos
-- ===============================
CREATE TABLE permisos (
  id_permiso INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  descripcion TEXT
);

-- ===============================
-- 3️⃣ Tabla: rol_permiso (relación muchos a muchos)
-- ===============================
CREATE TABLE rol_permiso (
  id_rol INT NOT NULL,
  id_permiso INT NOT NULL,
  PRIMARY KEY (id_rol, id_permiso),
  FOREIGN KEY (id_rol) REFERENCES rol(id_rol) ON DELETE CASCADE,
  FOREIGN KEY (id_permiso) REFERENCES permisos(id_permiso) ON DELETE CASCADE
);

-- ===============================
-- 4️⃣ Tabla: usuario
-- ===============================
CREATE TABLE usuario (
  id_usuario INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  apellido VARCHAR(100) NOT NULL,
  correo VARCHAR(100) UNIQUE,
  password VARCHAR(255) NOT NULL,
  id_rol INT,
  fecha_registro DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (id_rol) REFERENCES rol(id_rol) ON DELETE SET NULL
);

-- ===============================
-- 5️⃣ Tabla: tipo (tipo de habitación)
-- ===============================
CREATE TABLE tipo (
  id_tipo INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  capacidad INT NOT NULL,
  precio_base DECIMAL(10,2) NOT NULL,
  descripcion TEXT
);

-- ===============================
-- 6️⃣ Tabla: habitacion
-- ===============================
CREATE TABLE habitacion (
  id_habitacion INT AUTO_INCREMENT PRIMARY KEY,
  numero VARCHAR(10) UNIQUE NOT NULL,
  id_tipo INT NOT NULL,
  precio_total DECIMAL(10,2),
  piso INT,
  estado ENUM('disponible','ocupada','limpieza') DEFAULT 'disponible',
  descripcion TEXT,
  FOREIGN KEY (id_tipo) REFERENCES tipo(id_tipo) ON DELETE CASCADE
);

-- ===============================
-- 7️⃣ Tabla opcional: cliente
-- ===============================
CREATE TABLE cliente (
  id_cliente INT AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(100) NOT NULL,
  apellido VARCHAR(100) NOT NULL,
  ci VARCHAR(20) UNIQUE NOT NULL,
  correo VARCHAR(100),
  celular VARCHAR(20),
  direccion VARCHAR(150),
  fecha_registro DATETIME DEFAULT CURRENT_TIMESTAMP
);
